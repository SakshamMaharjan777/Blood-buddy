<script>
(async function () {
  const widths = [1280, 900, 768, 480, 430, 360];
  const results = [];

  function loadFrame(w) {
    return new Promise((resolve) => {
      const f = document.createElement('iframe');
      f.width = w;
      f.height = 900;
      f.src = '/CODE/HTML/landing.html';
      f.onload = () => resolve(f);
      document.getElementById('frame-wrap').appendChild(f);
    });
  }

  const getSel = (doc, s) => doc.querySelector(s);

  for (const w of widths) {
    const f = await loadFrame(w);
    await new Promise(r => setTimeout(r, 400)); // let fonts + JS settle
    const doc = f.contentDocument;
    const win = f.contentWindow;

    const m = { w, errors: [] };
    try {
      // 1. horizontal overflow check
      m.bodyScrollW = doc.body.scrollWidth;
      m.overflow = doc.body.scrollWidth > w;

      // 2. active media query state
      m.navLinksDisplay = win.getComputedStyle(getSel(doc, '#nav-links')).display;
      m.hamburgerDisplay = win.getComputedStyle(getSel(doc, '#hamburger')).display;

      // 3. grid column counts
      const grids = {
        stats: '.stats-grid', steps: '.steps-grid', roles: '.roles-grid',
        compat: '.compat-grid', searchRow: '.search-row'
      };
      m.grids = {};
      for (const [k, sel] of Object.entries(grids)) {
        const el = getSel(doc, sel);
        if (!el) { m.grids[k] = 'missing'; continue; }
        m.grids[k] = win.getComputedStyle(el).gridTemplateColumns.split(' ').length;
      }

      // 4. hamburger toggle works
      const ham = getSel(doc, '#hamburger');
      ham.click();
      await new Promise(r => setTimeout(r, 120));
      m.menuOpens = win.getComputedStyle(getSel(doc, '#nav-links')).display === 'flex';
      m.actionsOpen = win.getComputedStyle(getSel(doc, '#nav-actions')).display === 'flex';
      ham.click();
      await new Promise(r => setTimeout(r, 60));

      // 5. hero title fits inside frame
      const ht = getSel(doc, '.hero-title');
      m.heroTitleW = Math.round(ht.getBoundingClientRect().width);
      m.heroTitleFits = m.heroTitleW <= w;

      // 6. compat cards still 2-up at 480 (design intent) and readable
      m.compatCols480Intent = (w === 480) ? m.grids.compat : undefined;
    } catch (e) {
      m.errors.push(String(e));
    }
    results.push(m);
    f.remove();
  }

  window.__RESULTS__ = results;
  // human-readable dump
  const lines = [];
  for (const r of results) {
    lines.push(`=== width ${r.w} ===`);
    lines.push(`overflow=${r.overflow} bodyScrollW=${r.bodyScrollW}`);
    lines.push(`navLinks=${r.navLinksDisplay} hamburger=${r.hamburgerDisplay} menuOpens=${r.menuOpens} actionsOpen=${r.actionsOpen}`);
    lines.push(`grids=${JSON.stringify(r.grids)}`);
    lines.push(`heroTitleW=${r.heroTitleW} fits=${r.heroTitleFits}`);
    if (r.errors.length) lines.push(`ERRORS: ${r.errors.join(' | ')}`);
  }
  document.getElementById('out').textContent = lines.join('\n');
  document.title = 'RESPTEST-DONE';
})();
</script>
